public class Animal
    {
        public  virtual void EmitirSom()
        {
            Console.WriteLine("O animal faz o som");
        }
    }

public class Leao : Animal
    {
        public override void EmitirSom()
        {
            Console.WriteLine("O Leão faz: ROAR!!");
        }
    }

public class Macaco : Animal
    {
        public override  void EmitirSom()
        {
            Console.WriteLine("O macaco faz: UGH UGH AAH AAH");
        }
    }

public class Cobra : Animal
    {
        public override void EmitirSom()
        {
            Console.WriteLine("A cobra faz: Ssssssss");
        }
    }

List<Animal> animais = new List<Animal>();

            animais.Add(new Leao());
            animais.Add(new Macaco());
            animais.Add(new Cobra());

            foreach (Animal animal in animais)
            {
                animal.EmitirSom();
            }
