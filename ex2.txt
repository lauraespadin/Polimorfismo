public class Funcionario
    {
        public string Nome { get; set; }
        public double Salario { get; set; }

        public virtual double CalcularBonusAnual()
        {
            return Salario * 0.10;
        }
    }

public class Gerente : Funcionario
    {
        public override double CalcularBonusAnual()
        {
            return (Salario * 0.15) + 1000;
        }
    }

public class Desenvolvedor : Funcionario
    {
        public override double CalcularBonusAnual()
        {
            return Salario * 0.12;
        }
    }

 Funcionario funcionario = new Funcionario();
            funcionario.Nome = "José";
            funcionario.Salario = 3000;

            Funcionario gerente = new Gerente();
            gerente.Nome = "Sandra";
            gerente.Salario = 5000;

            Funcionario desenvolvedor = new Desenvolvedor();
            desenvolvedor.Nome = "Junior";
            desenvolvedor.Salario = 4000;

            Console.WriteLine($"Bônus do funcionário: R$ {funcionario.CalcularBonusAnual():F2}");
